(() => {
  const log=[]; let next=0, seq=0;
  const push=e=>{log.push({...e,seq:++seq,t:performance.now()});if(log.length>320)log.shift();};
  const target=e=>e.tagName.toLowerCase()+(e.id?'#'+e.id:'');
  const Native=window.ResizeObserver;
  window.ResizeObserver=class extends Native {
    constructor(callback) {
      const id=++next;
      const origin=Error().stack?.split('\n').slice(1,4).join('\n');
      super((entries,self)=>{
        push({kind:'deliver',id,entries:entries.map(e=>[target(e.target),e.contentRect.width,e.contentRect.height])});
        return callback(entries,self);
      });
      this.traceId=id;
      push({kind:'create',id,origin,name:callback.name});
    }
    observe(e,...opts) {
      push({kind:'observe',id:this.traceId,target:target(e)});
      return super.observe(e,...opts);
    }
  };
  const record=()=>new MutationObserver(entries=>{
    for(const e of entries){
      push({kind:'attribute',target:target(e.target),attribute:e.attributeName,value:e.target.getAttribute(e.attributeName)});
    }
  }).observe(document.documentElement,{subtree:true,attributes:true,attributeFilter:['style','class']});
  if(document.documentElement)record();else document.addEventListener('DOMContentLoaded',record,{once:true});
  addEventListener('error',e=>push({kind:'error',message:e.message,file:e.filename,line:e.lineno}));
  window.__roTrace={dump:()=>({log,errors:log.filter(e=>e.kind==='error')})};
})();
